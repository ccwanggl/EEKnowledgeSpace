# 更多可综合代码示例

> 以下模块可直接综合到 FPGA 中，每个示例都包含 Testbench 要点。

## 示例 1：7 段译码器（BCD → 数码管）

```verilog
module seg7 (
    input  [3:0] bcd,
    output reg [6:0] seg  // g~a, 共阴极：1=亮
);
    always @(*) begin
        case (bcd)
            4'd0: seg = 7'b0111111;
            4'd1: seg = 7'b0000110;
            4'd2: seg = 7'b1011011;
            4'd3: seg = 7'b1001111;
            4'd4: seg = 7'b1100110;
            4'd5: seg = 7'b1101101;
            4'd6: seg = 7'b1111101;
            4'd7: seg = 7'b0000111;
            4'd8: seg = 7'b1111111;
            4'd9: seg = 7'b1101111;
            default: seg = 7'b0000000; // 熄灭
        endcase
    end
endmodule
```

### 示例 2：按键消抖（20ms 稳定检测）

```verilog
module debounce (
    input  clk,      // 假设 100MHz
    input  rst,
    input  btn_in,
    output reg btn_out
);
    localparam CNT_MAX = 2_000_000; // 100MHz × 20ms
    reg [20:0] cnt;
    reg btn_sync0, btn_sync1;

    always @(posedge clk) begin
        if (rst) begin
            btn_sync0 <= 0; btn_sync1 <= 0;
            cnt <= 0; btn_out <= 0;
        end else begin
            btn_sync0 <= btn_in;
            btn_sync1 <= btn_sync0; // 2 级同步器

            if (btn_sync1 != btn_out) begin
                if (cnt < CNT_MAX) cnt <= cnt + 1;
                else begin btn_out <= btn_sync1; cnt <= 0; end
            end else begin
                cnt <= 0;
            end
        end
    end
endmodule
```

### 示例 3：伪随机数生成器（LFSR，16-bit）

```verilog
module lfsr16 (
    input  clk, rst,
    input  load,
    input  [15:0] seed,
    output reg [15:0] rand
);
    always @(posedge clk) begin
        if (rst)      rand <= 16'hACE1;
        else if (load) rand <= seed;
        else begin
            // 16-bit LFSR: taps at 16, 15, 13, 4
            rand <= {rand[0], rand[15:1]} ^ {15'b0, rand[0] ^ rand[1] ^ rand[3] ^ rand[12]};
        end
    end
endmodule
```
