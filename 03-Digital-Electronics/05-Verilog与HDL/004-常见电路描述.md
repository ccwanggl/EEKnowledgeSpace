# 常见电路描述

## 组合逻辑
```verilog
// 2-1 MUX
assign y = sel ? a : b;

// 4-1 MUX
always @(*) begin
    case(sel)
        2'b00: y = a;
        2'b01: y = b;
        2'b10: y = c;
        2'b11: y = d;
    endcase
end
```

### 时序逻辑：D 触发器
```verilog
always @(posedge clk or posedge reset) begin
    if (reset)
        q <= 1'b0;
    else
        q <= d;
end
```

### 计数器
```verilog
reg [7:0] count;
always @(posedge clk) begin
    if (reset)
        count <= 0;
    else if (enable)
        count <= count + 1;
end
```

### 状态机（Moore）
```verilog
reg [1:0] state, next_state;
localparam IDLE = 2'b00, RUN = 2'b01, STOP = 2'b10;

always @(posedge clk) begin
    if (reset) state <= IDLE;
    else state <= next_state;
end

always @(*) begin
    case (state)
        IDLE: next_state = start ? RUN : IDLE;
        RUN:  next_state = done ? STOP : RUN;
        STOP: next_state = IDLE;
        default: next_state = IDLE;
    endcase
end

assign output_signal = (state == RUN);
```
