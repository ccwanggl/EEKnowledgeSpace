# 字符设备驱动开发

用户态看到的驱动，本质上是一个文件：`open()` 建立会话、`read()/write()` 交换数据、`ioctl()` 传输控制命令。字符设备驱动就是实现这套文件语义的内核模块。本文写出一个最小但完整的内存字符设备，建立"驱动 ↔ 用户态"的完整数据通路。

## 三个核心对象

| 对象 | 作用 | 对应用户态 |
|------|------|-----------|
| `dev_t` 设备号 | 主设备号标识驱动，次设备号标识实例 | 文件节点背后的编号，`/proc/devices` 可查 |
| `struct cdev` | 把"操作集合 + 设备号"注册进内核 | 让 open 能找到你 |
| `struct file_operations` | 方法表：open/read/write/ioctl/release… | 用户态每个文件调用的落地处 |

`open()` 时内核用文件节点的设备号找到 `cdev`，此后该文件句柄的 read/write 都走同一个 `file_operations`。`struct file` 代表**一次打开**（每个 fd 一个实例），`private_data` 用来挂驱动的私有结构。

## 完整代码

```c
#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/device.h>
#include <linux/uaccess.h>

#define BUF_SIZE 256

struct mychar_dev {
    char buf[BUF_SIZE];
    size_t len;
    struct cdev cdev;
};

static dev_t devno;
static struct class *myclass;
static struct mychar_dev mydev;

static int my_open(struct inode *inode, struct file *filp)
{
    /* 从 cdev 反推私有结构，挂到本次打开的文件上 */
    filp->private_data = container_of(inode->i_cdev, struct mychar_dev, cdev);
    return 0;
}

static ssize_t my_read(struct file *filp, char __user *ubuf,
                       size_t count, loff_t *ppos)
{
    struct mychar_dev *dev = filp->private_data;
    ssize_t avail = dev->len - *ppos;      /* ppos：本次读写的文件位置 */

    if (avail <= 0)
        return 0;                          /* EOF */
    if (count > avail)
        count = avail;
    if (copy_to_user(ubuf, dev->buf + *ppos, count))
        return -EFAULT;
    *ppos += count;
    return count;
}

static ssize_t my_write(struct file *filp, const char __user *ubuf,
                        size_t count, loff_t *ppos)
{
    struct mychar_dev *dev = filp->private_data;

    if (count > BUF_SIZE)
        return -EINVAL;                    /* 简化：只支持整覆盖写 */
    if (copy_from_user(dev->buf, ubuf, count))
        return -EFAULT;
    dev->len = count;
    *ppos = 0;
    return count;
}

static struct file_operations my_fops = {
    .owner = THIS_MODULE,
    .open  = my_open,
    .read  = my_read,
    .write = my_write,
};

static int __init my_init(void)
{
    int ret;

    ret = alloc_chrdev_region(&devno, 0, 1, "mychar");   /* 动态申请主设备号 */
    if (ret)
        return ret;

    cdev_init(&mydev.cdev, &my_fops);
    mydev.cdev.owner = THIS_MODULE;
    mydev.len = 0;

    ret = cdev_add(&mydev.cdev, devno, 1);
    if (ret)
        goto fail_region;

    myclass = class_create(THIS_MODULE, "mychar");       /* ≥6.4 内核去掉第一个参数：class_create("mychar") */
    if (IS_ERR(myclass)) {
        ret = PTR_ERR(myclass);
        goto fail_cdev;
    }
    device_create(myclass, NULL, devno, NULL, "mychar"); /* devtmpfs 自动生成 /dev/mychar */
    pr_info("mychar: major=%d minor=%d\n", MAJOR(devno), MINOR(devno));
    return 0;

fail_cdev:
    cdev_del(&mydev.cdev);
fail_region:
    unregister_chrdev_region(devno, 1);
    return ret;
}

static void __exit my_exit(void)
{
    device_destroy(myclass, devno);
    class_destroy(myclass);
    cdev_del(&mydev.cdev);
    unregister_chrdev_region(devno, 1);
}

module_init(my_init);
module_exit(my_exit);
MODULE_LICENSE("GPL");
MODULE_DESCRIPTION("minimal char device");
```

要点：

- **申请资源顺序 = 释放的逆序**。`alloc_chrdev_region → cdev_add → class_create → device_create`，退出时反过来拆，任何一步失败回滚已获资源。内核资源是全局命名空间，泄漏一次就可能让下次加载失败。
- `alloc_chrdev_region` 从内核动态分配主设备号（避免和固定号段冲突）；固定编号的旧驱动用 `register_chrdev_region`。
- `device_create` + devtmpfs 才有 `/dev/mychar` 节点；不带 class 时只能手工 `mknod /dev/mychar c <major> 0`（major 从 `dmesg` 或 `/proc/devices` 读）。

## 为什么必须 copy_to_user

用户态指针在内核态**不能直接解引用**：该地址对应的页在内核上下文可能未映射（比如用户内存被换出、或属于另一个进程），直接解引用轻则读到错误数据，重则内核 oops。`copy_to_user/copy_from_user` 做了三件事：访问权限检查、缺页处理、越界防护，返回未完成的字节数（非 0 即 `-EFAULT`）。`__user` 注解是给 sparse 检查器看的，提醒这个指针来自用户态。

## ioctl：不适合读写数据时

read/write 是字节流；设置采样率、读取固件版本这类**控制语义**走 `unlocked_ioctl`：

```c
#define MYCHAR_IOC_MAGIC 'm'
#define MYCHAR_IOC_GET_LEN _IOR(MYCHAR_IOC_MAGIC, 1, size_t)
#define MYCHAR_IOC_CLR     _IO (MYCHAR_IOC_MAGIC, 2)

static long my_ioctl(struct file *filp, unsigned int cmd, unsigned long arg)
{
    struct mychar_dev *dev = filp->private_data;

    switch (cmd) {
    case MYCHAR_IOC_GET_LEN:
        if (copy_to_user((void __user *)arg, &dev->len, sizeof(dev->len)))
            return -EFAULT;
        return 0;
    case MYCHAR_IOC_CLR:
        dev->len = 0;
        return 0;
    default:
        return -ENOTTY;     /* 内核约定：不支持的 ioctl 命令返回 ENOTTY */
    }
}
```

命令字用 `_IO/_IOR/_IOW` 宏编码（方向、序号、数据类型），用户态和驱动用同一个头文件共享定义。不支持的命令返回 `-ENOTTY` 而不是 `-EINVAL`，使 `errno` 语义与内核惯例一致。

## 用户态测试程序

```c
#include <fcntl.h>
#include <stdio.h>
#include <string.h>
#include <unistd.h>

int main(void)
{
    int fd = open("/dev/mychar", O_RDWR);
    if (fd < 0) { perror("open"); return 1; }

    const char *msg = "hello from user space";
    if (write(fd, msg, strlen(msg)) < 0) { perror("write"); return 1; }

    char buf[64] = {0};
    ssize_t n = read(fd, buf, sizeof(buf));   /* ppos 由 write 归 0 */
    printf("read %zd bytes: %s\n", n, buf);
    close(fd);
    return 0;
}
```

设备节点默认属主 root，测试程序需 root 运行；量产环境用 udev 规则改属主/权限。

## 常见陷阱

- **忘记处理 `ppos`**：read/write 的文件偏移完全由驱动负责，忽略它会导致第二次读不到数据或永远读到同一处。
- **并发**：本例是单缓冲玩具。真实驱动多个进程同时 open 时，`buf/len` 的读写需要锁（mutex/spinlock）保护——先想清楚"哪些状态属于设备、哪些属于一次打开"。
- **卸载时句柄未关闭**：`rmmod` 前若有进程持有 fd，内核会因 `.owner = THIS_MODULE` 引用计数拒绝卸载（good）；没有 owner 的模块卸载瞬间已打开的 fd 调用你的函数 → oops。
- **`class_create` 签名随版本变化**：6.4+ 内核去掉 `THIS_MODULE` 参数，编译报错时先核对该内核版本的 API。

## 检查清单

- 设备号申请/释放配对，cdev/class/device 创建/销毁配对。
- 所有用户态指针经过 `copy_to_user/from_user`，返回值检查。
- 不支持的用户态输入返回明确的负 errno。
- `file_operations.owner = THIS_MODULE` 不省略。

## 实验

1. 编译加载模块，`cat /proc/devices | grep mychar` 确认主设备号，`ls -l /dev/mychar` 确认节点生成。
2. 编译运行用户态测试程序，观察读写闭环。
3. 增加一个 `_IOW` 命令（如向设备写入一个 u32 配置），用户态验证。
